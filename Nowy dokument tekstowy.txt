#!/bin/bash

DRY_RUN=true   # <<< ZMIEN NA false gdy gotowy

BACKUP="/c/Users/admin_test/Desktop/hook/3dstudiopoland-bot/1BULK_Backup"


BATCH=100
MAX_BATCHES=10
SLEEP=120

COUNTER_FILE=".commit_counter"
PROGRESS_FILE=".progress"

mkdir -p "$BACKUP"

if [ ! -f "$COUNTER_FILE" ]; then
  echo 1 > "$COUNTER_FILE"
fi

GLOBAL_COUNT=$(cat "$COUNTER_FILE")

git ls-files --others --exclude-standard > all_files.txt

TOTAL=$(wc -l < all_files.txt)

if [ "$TOTAL" -lt "$BATCH" ]; then
  echo "Only $TOTAL files left. Less than $BATCH. Stopping."
  rm all_files.txt
  exit 0
fi

split -l $BATCH all_files.txt batch_

START_BATCH=1
if [ -f "$PROGRESS_FILE" ]; then
  START_BATCH=$(cat "$PROGRESS_FILE")
  echo "Resuming from batch $START_BATCH"
fi

COUNT=1

for f in batch_*; do

  if [ $COUNT -lt $START_BATCH ]; then
    ((COUNT++))
    continue
  fi

  if [ $COUNT -gt $MAX_BATCHES ]; then
    echo "Reached max batches ($MAX_BATCHES)."
    break
  fi

  FILES_IN_BATCH=$(wc -l < "$f")

  if [ "$FILES_IN_BATCH" -lt "$BATCH" ]; then
    echo "Last batch has only $FILES_IN_BATCH files. Stopping."
    break
  fi

  echo "======== DRY RUN Batch $COUNT ========"

  cat "$f"

  if [ "$DRY_RUN" = true ]; then
    echo "DRY RUN: skipping backup/git/sleep"
    ((COUNT++))
    continue
  fi

  BDIR="$BACKUP/batch_$GLOBAL_COUNT"
  mkdir -p "$BDIR"

  while read file; do
    mkdir -p "$BDIR/$(dirname "$file")"
    cp "$file" "$BDIR/$file"
  done < "$f"

  xargs git add < "$f"
  git commit -m "Auto batch $GLOBAL_COUNT"
  git push || exit 1

  echo $((COUNT+1)) > "$PROGRESS_FILE"

  GLOBAL_COUNT=$((GLOBAL_COUNT+1))
  echo "$GLOBAL_COUNT" > "$COUNTER_FILE"

  sleep $SLEEP

  ((COUNT++))

done

rm batch_* all_files.txt

if [ "$DRY_RUN" = false ]; then
  rm "$PROGRESS_FILE"
fi

echo "Finished."
